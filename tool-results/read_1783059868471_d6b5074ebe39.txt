    22→// Fade-in on scroll animation wrapper
    23→function FadeInWhenVisible({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    24→  const ref = useRef(null)
    25→  const isInView = useInView(ref, { once: true, margin: '-50px' })
    26→
    27→  return (
    28→    <motion.div
    29→      ref={ref}
    30→      initial={{ opacity: 0, y: 30 }}
    31→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    32→      transition={{ duration: 0.7, delay, ease: [0.25, 0.4, 0.25, 1] }}
    33→      className={className}
    34→    >
    35→      {children}
    36→    </motion.div>
    37→  )
    38→}

    40→// Navigation Bar
    41→function NavBar({ onMasuk }: { onMasuk: () => void }) {
    42→  const [scrolled, setScrolled] = useState(false)
    43→  const [mobileOpen, setMobileOpen] = useState(false)
    44→
    45→  useEffect(() => {
    46→    const handleScroll = () => setScrolled(window.scrollY > 20)
    47→    window.addEventListener('scroll', handleScroll)
    48→    return () => window.removeEventListener('scroll', handleScroll)
    49→  }, [])
    50→
    51→  const navLinks = [
    52→    { label: 'Tentang', href: '#about' },
    53→    { label: 'Layanan', href: '#layanan' },
    54→    { label: 'Insights', href: '#insights' },
    55→    { label: 'Karir', href: '#karir' },
    56→    { label: 'Kontak', href: '#kontak' },
    57→  ]
    58→
    59→  const handleNavClick = (href: string) => {
    60→    setMobileOpen(false)
    61→    const el = document.querySelector(href)
    62→    if (el) el.scrollIntoView({ behavior: 'smooth' })
    63→  }
    64→
    65→  return (
    66→    <header
    67→      className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
    68→        scrolled
    69→          ? 'bg-white/95 backdrop-blur-md shadow-sm border-b border-[#e5e0d5]'
    70→          : 'bg-transparent'
    71→      }`}
    72→    >
    73→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
    74→        <div className="flex items-center justify-between h-16 lg:h-18">
    75→          {/* Logo */}
    76→          <a href="#" className="flex items-center gap-2" onClick={(e) => { e.preventDefault(); window.scrollTo({ top: 0, behavior: 'smooth' }) }}>
    77→            <div className="w-8 h-8 rounded-sm bg-[#0D503C] flex items-center justify-center">
    78→              <span className="text-white text-xs font-bold tracking-wider">JAC</span>
    79→            </div>
    80→            <span className="hidden sm:block text-[#0D503C] font-semibold text-sm tracking-wide">
    81→              Jiwa Abdi Cinta
    82→            </span>
    83→          </a>
    84→
    85→          {/* Desktop Nav */}
    86→          <nav className="hidden lg:flex items-center gap-8">
    87→            {navLinks.map((link) => (
    88→              <button
    89→                key={link.href}
    90→                onClick={() => handleNavClick(link.href)}
    91→                className="text-[#3a3a4a] hover:text-[#0D503C] text-sm font-medium transition-colors relative group"
    92→              >
    93→                {link.label}
    94→                <span className="absolute -bottom-1 left-0 w-0 h-0.5 bg-[#0D503C] transition-all duration-300 group-hover:w-full" />
    95→              </button>
    96→            ))}
    97→          </nav>
    98→
    99→          {/* CTA + Mobile Toggle */}
   100→          <div className="flex items-center gap-3">
   101→            <button
   102→              onClick={onMasuk}
   103→              className="hidden sm:inline-flex items-center gap-2 px-5 py-2 bg-[#0D503C] text-white text-sm font-medium rounded-sm hover:bg-[#0a3f2f] transition-colors"
   104→            >
   105→              Masuk
   106→            </button>
   107→            <button
   108→              onClick={() => setMobileOpen(!mobileOpen)}
   109→              className="lg:hidden p-2 text-[#3a3a4a] hover:text-[#0D503C] transition-colors"
   110→              aria-label="Toggle menu"
   111→            >
   112→              {mobileOpen ? <X className="w-5 h-5" /> : <Menu className="w-5 h-5" />}
   113→            </button>
   114→          </div>
   115→        </div>
   116→      </div>
   117→
   118→      {/* Mobile Menu */}
   119→      <AnimatePresence>
   120→        {mobileOpen && (
   121→          <motion.div
   122→            initial={{ opacity: 0, height: 0 }}
   123→            animate={{ opacity: 1, height: 'auto' }}
   124→            exit={{ opacity: 0, height: 0 }}
   125→            transition={{ duration: 0.3 }}
   126→            className="lg:hidden bg-white border-b border-[#e5e0d5] overflow-hidden"
   127→          >
   128→            <div className="px-4 py-4 space-y-1">
   129→              {navLinks.map((link) => (
   130→                <button
   131→                  key={link.href}
   132→                  onClick={() => handleNavClick(link.href)}
   133→                  className="block w-full text-left px-3 py-2.5 text-[#3a3a4a] hover:text-[#0D503C] hover:bg-[#F5F0E8] text-sm font-medium rounded transition-colors"
   134→                >
   135→                  {link.label}
   136→                </button>
   137→              ))}
   138→              <div className="pt-3 border-t border-[#e5e0d5]">
   139→                <button
   140→                  onClick={() => { setMobileOpen(false); onMasuk() }}
   141→                  className="w-full px-4 py-2.5 bg-[#0D503C] text-white text-sm font-medium rounded-sm hover:bg-[#0a3f2f] transition-colors"
   142→                >
   143→                  Masuk Dashboard
   144→                </button>
   145→              </div>
   146→            </div>
   147→          </motion.div>
   148→        )}
   149→      </AnimatePresence>
   150→    </header>
   151→  )
   152→}
