    52→    52→/** Extract up to two initials from a name, stripping common titles. */
    53→    53→function getInitials(name: string): string {
    54→    54→  const cleaned = name
    55→    55→    .replace(/,/g, ' ')
    56→    56→    .replace(/\b(SE|S\.H\.|M\.Pd|Dr\.|Ir\.|M\.M|Ph\.D)\b/g, ' ')
    57→    57→  const parts = cleaned.split(/\s+/).filter(p => p.length > 1 && /^[A-Za-z]/.test(p))
    58→    58→  return parts.slice(0, 2).map(p => p[0]?.toUpperCase() ?? '').join('')
    59→    59→}
