    52→/** Extract up to two initials from a name, stripping common titles. */
    53→function getInitials(name: string): string {
    54→  const cleaned = name
    55→    .replace(/,/g, ' ')
    56→    .replace(/\b(SE|S\.H\.|M\.Pd|Dr\.|Ir\.|M\.M|Ph\.D)\b/g, ' ')
    57→  const parts = cleaned.split(/\s+/).filter(p => p.length > 1 && /^[A-Za-z]/.test(p))
    58→  return parts.slice(0, 2).map(p => p[0]?.toUpperCase() ?? '').join('')
    59→}

    89→/** A single member card. Optionally tinted with the dept color (Kepala Divisi). */
    90→function OrgMemberCard({ member, withDept }: { member: OrgMember; withDept?: boolean }) {
    91→  const deptColor = withDept && member.dept ? DEPT_COLORS[member.dept] : null
    92→  return (
    93→    <div className="bg-white border border-[#D4E0D4] rounded-sm p-5 relative overflow-hidden h-full hover:shadow-md hover:border-[#2D5A3D]/30 transition-all">
    94→      {deptColor && (
    95→        <div className="absolute top-0 left-0 right-0 h-1" style={{ backgroundColor: deptColor }} aria-hidden="true" />
    96→      )}
    97→      <div className={`flex items-start gap-3 ${deptColor ? 'mt-1.5' : ''}`}>
    98→        {deptColor ? (
    99→          <div
   100→            className="w-9 h-9 rounded-sm flex items-center justify-center flex-shrink-0"
   101→            style={{ backgroundColor: `${deptColor}18`, color: deptColor }}
   102→            aria-hidden="true"
   103→          >
   104→            <Building2 className="w-4 h-4" />
   105→          </div>
   106→        ) : (
   107→          <div className="w-9 h-9 rounded-sm bg-[#2D5A3D]/10 text-[#2D5A3D] flex items-center justify-center flex-shrink-0" aria-hidden="true">
   108→            <Building2 className="w-4 h-4" />
   109→          </div>
   110→        )}
   111→        <div className="min-w-0">

   120→/** A row of member cards with horizontal + vertical connectors (org chart style). */
   121→function OrgRow({ members, withDept, layout }: {
   122→  members: OrgMember[]
   123→  withDept?: boolean
   124→  layout: 'three' | 'five'
   125→}) {
   126→  const gridCls = layout === 'three'
   127→    ? 'grid-cols-1 lg:grid-cols-3'
   128→    : 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-5'
   129→  const barCls = layout === 'three'
   130→    ? 'lg:left-[16.6667%] lg:right-[16.6667%]'
   131→    : 'lg:left-[10%] lg:right-[10%]'
   132→
   133→  return (
   134→    <div className="relative">
   135→      {/* Horizontal connector bar (visible only on lg for a clean org-chart look) */}
   136→      <div
   137→        className={`hidden lg:block absolute top-0 h-px bg-[#2D5A3D]/30 ${barCls}`}
   138→        aria-hidden="true"
   139→      />
   140→      <div className={`grid gap-4 ${gridCls}`}>
   141→        {members.map((m, i) => (
   142→          <div key={i} className="flex flex-col items-center">
   143→            {/* Vertical drop above each card */}
   144→            <div className="w-px h-7 bg-[#2D5A3D]/30" aria-hidden="true" />
   145→            <div className="w-full">
   146→              <OrgMemberCard member={m} withDept={withDept} />
   147→            </div>
   148→          </div>
   149→        ))}
   150→      </div>
   151→    </div>
   152→  )
   153→}
