    22→    22→// Fade-in on scroll animation wrapper
    23→    23→function FadeInWhenVisible({ children, delay = 0, className = '' }: { children: React.ReactNode; delay?: number; className?: string }) {
    24→    24→  const ref = useRef(null)
    25→    25→  const isInView = useInView(ref, { once: true, margin: '-50px' })
    26→    26→
    27→    27→  return (
    28→    28→    <motion.div
    29→    29→      ref={ref}
    30→    30→      initial={{ opacity: 0, y: 30 }}
    31→    31→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    32→    32→      transition={{ duration: 0.7, delay, ease: [0.25, 0.4, 0.25, 1] }}
    33→    33→      className={className}
    34→    34→    >
    35→    35→      {children}
    36→    36→    </motion.div>
    37→    37→  )
    38→    38→}
