    29→/* ============================================================
    30→   SEEDED PRNG (mulberry32) — deterministic so server & client
    31→   render identical particles (prevents hydration mismatch).
    32→   ============================================================ */
    33→function mulberry32(seed: number) {
    34→  return function () {
    35→    seed |= 0
    36→    seed = (seed + 0x6D2B79F5) | 0
    37→    let t = seed
    38→    t = Math.imul(t ^ (t >>> 15), t | 1)
    39→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    40→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    41→  }
    42→}

    44→/* ============================================================
    45→   SAGE PARTICLE FIELD — soft floating dots on light bg
    46→   Uses seeded PRNG so SSR & client match (no hydration error).
    47→   ============================================================ */
    48→function SageParticles() {
    49→  const particles = useMemo(() => {
    50→    const rand = mulberry32(20240421) // fixed seed
    51→    return Array.from({ length: 24 }).map((_, i) => ({
    52→      id: i,
    53→      x: rand() * 100,
    54→      y: rand() * 100,
    55→      size: rand() * 4 + 2,
    56→      duration: rand() * 8 + 6,
    57→      delay: rand() * 5,
    58→      opacity: rand() * 0.4 + 0.15,
    59→      isGold: rand() > 0.5,
    60→    }))
    61→  }, [])
    62→  return (
    63→    <div className="absolute inset-0 overflow-hidden pointer-events-none" aria-hidden>
    64→      {particles.map(p => (
    65→        <motion.span
    66→          key={p.id}

    72→              ? 'radial-gradient(circle, #E8D9A0, #C9A961)'

    89→/* ============================================================
    90→   ANIMATED COUNTER — counts up when in view
    91→   ============================================================ */
    92→function Counter({ value, suffix = '', duration = 2 }: { value: number; suffix?: string; duration?: number }) {
    93→  const ref = useRef<HTMLSpanElement>(null)
    94→  const inView = useInView(ref, { once: true, margin: '-50px' })
    95→  const [display, setDisplay] = useState(0)
    96→
    97→  useEffect(() => {
    98→    if (!inView) return
    99→    let raf = 0
   100→    const start = performance.now()
   101→    const tick = (now: number) => {
   102→      const t = Math.min((now - start) / (duration * 1000), 1)
   103→      const eased = 1 - Math.pow(1 - t, 3)
   104→      setDisplay(Math.round(eased * value))
   105→      if (t < 1) raf = requestAnimationFrame(tick)
   106→    }
   107→    raf = requestAnimationFrame(tick)
   108→    return () => cancelAnimationFrame(raf)
   109→  }, [inView, value, duration])
   110→
   111→  return <span ref={ref}>{display}{suffix}</span>
   112→}

   114→/* ============================================================
   115→   3D TILT CARD — mouse-following tilt
   116→   ============================================================ */
   117→function TiltCard({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   118→  const ref = useRef<HTMLDivElement>(null)
   119→  const [transform, setTransform] = useState('')
   120→  const handleMove = (e: React.MouseEvent) => {
   121→    const el = ref.current
   122→    if (!el) return
   123→    const rect = el.getBoundingClientRect()
   124→    const x = (e.clientX - rect.left) / rect.width - 0.5
   125→    const y = (e.clientY - rect.top) / rect.height - 0.5
   126→    setTransform(`perspective(1000px) rotateY(${x * 8}deg) rotateX(${-y * 8}deg) translateZ(8px)`)
   127→  }
   128→  const handleLeave = () => setTransform('perspective(1000px) rotateY(0) rotateX(0) translateZ(0)')
   129→  return (
   130→    <div
   131→      ref={ref}
   132→      onMouseMove={handleMove}
   133→      onMouseLeave={handleLeave}
   134→      style={{ transform, transition: 'transform 0.2s ease-out', transformStyle: 'preserve-3d' }}
   135→      className={className}
   136→    >
   137→      {children}
   138→    </div>
   139→  )
   140→}
