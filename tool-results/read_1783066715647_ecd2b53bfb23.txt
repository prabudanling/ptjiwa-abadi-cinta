    11→/* ============================================================
    12→   ANIMATED COUNTER — counts up when in view, with prefix + suffix
    13→   ============================================================ */
    14→function Counter({
    15→  value, prefix = '', suffix = '', duration = 2,
    16→}: { value: number; prefix?: string; suffix?: string; duration?: number }) {
    17→  const ref = useRef<HTMLSpanElement>(null)
    18→  const inView = useInView(ref, { once: true, margin: '-50px' })
    19→  const [display, setDisplay] = useState(0)
    20→
    21→  useEffect(() => {
    22→    if (!inView) return
    23→    let raf = 0
    24→    const start = performance.now()
    25→    const tick = (now: number) => {
    26→      const t = Math.min((now - start) / (duration * 1000), 1)
    27→      const eased = 1 - Math.pow(1 - t, 3)
    28→      setDisplay(Math.round(eased * value))
    29→      if (t < 1) raf = requestAnimationFrame(tick)
    30→    }
    31→    raf = requestAnimationFrame(tick)
    32→    return () => cancelAnimationFrame(raf)
    33→  }, [inView, value, duration])
    34→
    35→  return <span ref={ref}>{prefix}{display}{suffix}</span>
    36→}

    41→function FadeIn({
    42→  children, delay = 0, className = '',
    43→}: { children: React.ReactNode; delay?: number; className?: string }) {
    44→  const ref = useRef(null)
    45→  const inView = useInView(ref, { once: true, margin: '-50px' })
    46→  return (
    47→    <motion.div
    48→      ref={ref}
    49→      initial={{ opacity: 0, y: 30 }}
    50→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    51→      transition={{ duration: 0.7, delay, ease: [0.25, 0.4, 0.25, 1] }}
    52→      className={className}
    53→    >
    54→      {children}
    55→    </motion.div>
    56→  )
    57→}
