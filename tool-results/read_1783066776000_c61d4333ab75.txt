    19→/* ============================================================
    20→   STAT PARSER — extract prefix/number/suffix from stat strings
    21→   like "340+", "Rp 47M", "5.200 ha", "4.8★", "1.250 t/h"
    22→   ============================================================ */
    23→type ParsedStat = {
    24→  prefix: string
    25→  num: number | null
    26→  suffix: string
    27→  format: (n: number) => string
    28→  raw: string
    29→}
    30→
    31→function parseStat(value: string): ParsedStat {
    32→  const m = value.match(/^(\D*)(\d[\d\.,]*)(.*)$/)
    33→  if (!m) return { prefix: '', num: null, suffix: '', format: () => value, raw: value }
    34→  const [, prefix, numStr, suffix] = m
    35→
    36→  // Indonesian thousand-separated integer: "5.200", "1.250", "10.000"
    37→  const isThousandSeparated = /^\d{1,3}(\.\d{3})+$/.test(numStr)
    38→  if (isThousandSeparated) {
    39→    const num = parseInt(numStr.replace(/\./g, ''), 10)
    40→    return {
    41→      prefix, num, suffix,
    42→      format: (n) => Math.round(n).toLocaleString('id-ID'),
    43→      raw: value,
    44→    }
    45→  }
    46→
    47→  // Comma decimal: "4,8"
    48→  if (numStr.includes(',')) {
    49→    const num = parseFloat(numStr.replace(',', '.'))
    50→    const decimals = (numStr.split(',')[1] || '').length
    51→    return {
    52→      prefix, num, suffix,
    53→      format: (n) => n.toFixed(decimals).replace('.', ','),
    54→      raw: value,
    55→    }
    56→  }
    57→
    58→  // Dot decimal: "4.8", "14.85"
    59→  const num = parseFloat(numStr)
    60→  const decimals = (numStr.split('.')[1] || '').length
    61→  return {
    62→    prefix, num, suffix,
    63→    format: (n) => decimals > 0 ? n.toFixed(decimals) : Math.round(n).toString(),
    64→    raw: value,
    65→  }
    66→}

    68→/* ============================================================
    69→   ANIMATED COUNTER — counts up when in view
    70→   ============================================================ */
    71→function Counter({ value, duration = 1.8 }: { value: string; duration?: number }) {
    72→  const ref = useRef<HTMLSpanElement>(null)
    73→  const inView = useInView(ref, { once: true, margin: '-50px' })
    74→  const parsed = useMemo(() => parseStat(value), [value])
    75→  const [display, setDisplay] = useState<string>(parsed.num !== null ? parsed.format(0) : parsed.raw)
    76→
    77→  useEffect(() => {
    78→    const target = parsed.num
    79→    if (target === null || !inView) return
    80→    let raf = 0
    81→    const start = performance.now()
    82→    const tick = (now: number) => {
    83→      const t = Math.min((now - start) / (duration * 1000), 1)
    84→      const eased = 1 - Math.pow(1 - t, 3)
    85→      setDisplay(parsed.format(eased * target))
    86→      if (t < 1) raf = requestAnimationFrame(tick)
    87→    }
    88→    raf = requestAnimationFrame(tick)
    89→    return () => cancelAnimationFrame(raf)
    90→  }, [inView, parsed, duration])
    91→
    92→  if (parsed.num === null) return <span ref={ref}>{parsed.raw}</span>
    93→  return <span ref={ref}>{parsed.prefix}{display}{parsed.suffix}</span>
    94→}

    99→function FadeIn({
   100→  children, delay = 0, className = '',
   101→}: { children: React.ReactNode; delay?: number; className?: string }) {
   102→  const ref = useRef(null)
   103→  const inView = useInView(ref, { once: true, margin: '-50px' })

   105→    <motion.div
   106→      ref={ref}
   107→      initial={{ opacity: 0, y: 30 }}
   108→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
   109→      transition={{ duration: 0.7, delay, ease: [0.25, 0.4, 0.25, 1] }}
   110→      className={className}
   111→    >
   112→      {children}
   113→    </motion.div>
   114→  )
   115→}

   117→/* ============================================================
   118→   GOLD PARTICLES — luxury floating dots
   119→   ============================================================ */
   120→function GoldParticles({ count = 22 }: { count?: number }) {
   121→  const particles = useMemo(() => Array.from({ length: count }).map((_, i) => ({
   122→    id: i,
   123→    x: Math.random() * 100,
   124→    y: Math.random() * 100,
   125→    size: Math.random() * 2.5 + 1,
   126→    duration: Math.random() * 8 + 6,
   127→    delay: Math.random() * 5,
   128→    opacity: Math.random() * 0.5 + 0.2,
   129→  })), [count])
   130→  return (
   131→    <div className="absolute inset-0 overflow-hidden pointer-events-none" aria-hidden>
   132→      {particles.map(p => (
   133→        <motion.span
   134→          key={p.id}
   135→          className="absolute rounded-full"
   136→          style={{
   137→            left: `${p.x}%`, top: `${p.y}%`,
   138→            width: p.size, height: p.size,
   139→            background: 'radial-gradient(circle, #F0E4B8, #C9A961)',
   140→            boxShadow: '0 0 6px rgba(245,208,97,0.6)',
   141→          }}
   142→          animate={{ y: [0, -30, 0], opacity: [p.opacity, p.opacity * 1.8, p.opacity] }}
   143→          transition={{ duration: p.duration, delay: p.delay, repeat: Infinity, ease: 'easeInOut' }}
   144→        />
   145→      ))}
   146→    </div>
   147→  )
   148→}
