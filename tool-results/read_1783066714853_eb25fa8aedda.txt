    19→    19→/* ============================================================
    20→    20→   STAT PARSER — extract prefix/number/suffix from stat strings
    21→    21→   like "340+", "Rp 47M", "5.200 ha", "4.8★", "1.250 t/h"
    22→    22→   ============================================================ */
    23→    23→type ParsedStat = {
    24→    24→  prefix: string
    25→    25→  num: number | null
    26→    26→  suffix: string
    27→    27→  format: (n: number) => string
    28→    28→  raw: string
    29→    29→}
    30→    30→
    31→    31→function parseStat(value: string): ParsedStat {
    32→    32→  const m = value.match(/^(\D*)(\d[\d\.,]*)(.*)$/)
    33→    33→  if (!m) return { prefix: '', num: null, suffix: '', format: () => value, raw: value }
    34→    34→  const [, prefix, numStr, suffix] = m
    35→    35→
    36→    36→  // Indonesian thousand-separated integer: "5.200", "1.250", "10.000"
    37→    37→  const isThousandSeparated = /^\d{1,3}(\.\d{3})+$/.test(numStr)
    38→    38→  if (isThousandSeparated) {
    39→    39→    const num = parseInt(numStr.replace(/\./g, ''), 10)
    40→    40→    return {
    41→    41→      prefix, num, suffix,
    42→    42→      format: (n) => Math.round(n).toLocaleString('id-ID'),
    43→    43→      raw: value,
    44→    44→    }
    45→    45→  }
    46→    46→
    47→    47→  // Comma decimal: "4,8"
    48→    48→  if (numStr.includes(',')) {
    49→    49→    const num = parseFloat(numStr.replace(',', '.'))
    50→    50→    const decimals = (numStr.split(',')[1] || '').length
    51→    51→    return {
    52→    52→      prefix, num, suffix,
    53→    53→      format: (n) => n.toFixed(decimals).replace('.', ','),
    54→    54→      raw: value,
    55→    55→    }
    56→    56→  }
    57→    57→
    58→    58→  // Dot decimal: "4.8", "14.85"
    59→    59→  const num = parseFloat(numStr)
    60→    60→  const decimals = (numStr.split('.')[1] || '').length
    61→    61→  return {
    62→    62→    prefix, num, suffix,
    63→    63→    format: (n) => decimals > 0 ? n.toFixed(decimals) : Math.round(n).toString(),
    64→    64→    raw: value,
    65→    65→  }
    66→    66→}

    68→    68→/* ============================================================
    69→    69→   ANIMATED COUNTER — counts up when in view
    70→    70→   ============================================================ */
    71→    71→function Counter({ value, duration = 1.8 }: { value: string; duration?: number }) {
    72→    72→  const ref = useRef<HTMLSpanElement>(null)
    73→    73→  const inView = useInView(ref, { once: true, margin: '-50px' })
    74→    74→  const parsed = useMemo(() => parseStat(value), [value])
    75→    75→  const [display, setDisplay] = useState<string>(parsed.num !== null ? parsed.format(0) : parsed.raw)
    76→    76→
    77→    77→  useEffect(() => {
    78→    78→    const target = parsed.num
    79→    79→    if (target === null || !inView) return
    80→    80→    let raf = 0
    81→    81→    const start = performance.now()
    82→    82→    const tick = (now: number) => {
    83→    83→      const t = Math.min((now - start) / (duration * 1000), 1)
    84→    84→      const eased = 1 - Math.pow(1 - t, 3)
    85→    85→      setDisplay(parsed.format(eased * target))
    86→    86→      if (t < 1) raf = requestAnimationFrame(tick)
    87→    87→    }
    88→    88→    raf = requestAnimationFrame(tick)
    89→    89→    return () => cancelAnimationFrame(raf)
    90→    90→  }, [inView, parsed, duration])
    91→    91→
    92→    92→  if (parsed.num === null) return <span ref={ref}>{parsed.raw}</span>
    93→    93→  return <span ref={ref}>{parsed.prefix}{display}{parsed.suffix}</span>
    94→    94→}
